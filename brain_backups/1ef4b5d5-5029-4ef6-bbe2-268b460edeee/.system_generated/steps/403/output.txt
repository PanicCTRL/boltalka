function PushPrice(high, low)
    if not high or high == 0 or not low or low == 0 then return end

    table.insert(maxPrices, high)
    if #maxPrices > 5 then
        table.remove(maxPrices, 1)
    end

    table.insert(minPrices, low)
    if #minPrices > 5 then
        table.remove(minPrices, 1)
    end
end
